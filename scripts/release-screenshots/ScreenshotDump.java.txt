    private static final LegacyComponentSerializer LEGACY = LegacyComponentSerializer.builder().character('§').hexColors().useUnusualXRepeatedCharacterHexFormat().build();
    private final List<Map<String, Object>> out = new ArrayList<>();

    private void dump(String name, PlayerMock p) {
        Inventory inv = p.getOpenInventory().getTopInventory();
        Map<String, Object> sc = new LinkedHashMap<>();
        sc.put("name", name);
        sc.put("title", LEGACY.serialize(p.getOpenInventory().title()));
        sc.put("rows", inv.getSize() / 9);
        Map<String, Object> slots = new LinkedHashMap<>();
        for (int i = 0; i < inv.getSize(); i++) {
            ItemStack s = inv.getItem(i);
            if (s == null || s.getType().isAir()) {
                continue;
            }
            Map<String, Object> it = new LinkedHashMap<>();
            it.put("m", s.getType().name());
            it.put("n", s.getAmount());
            var meta = s.getItemMeta();
            it.put("name", meta != null && meta.displayName() != null ? LEGACY.serialize(meta.displayName()) : s.getType().name());
            List<String> lore = new ArrayList<>();
            if (meta != null && meta.lore() != null) {
                for (Component c : meta.lore()) {
                    lore.add(LEGACY.serialize(c));
                }
            }
            it.put("lore", lore);
            slots.put(String.valueOf(i), it);
        }
        sc.put("slots", slots);
        out.add(sc);
    }

    private static String json(Object o) {
        if (o instanceof Map<?, ?> m) {
            StringBuilder sb = new StringBuilder("{");
            boolean first = true;
            for (var e : m.entrySet()) {
                sb.append(first ? "" : ",").append(json(String.valueOf(e.getKey()))).append(':').append(json(e.getValue()));
                first = false;
            }
            return sb.append('}').toString();
        }
        if (o instanceof List<?> l) {
            StringBuilder sb = new StringBuilder("[");
            for (int i = 0; i < l.size(); i++) {
                sb.append(i == 0 ? "" : ",").append(json(l.get(i)));
            }
            return sb.append(']').toString();
        }
        if (o instanceof Number || o instanceof Boolean) {
            return o.toString();
        }
        return "\"" + String.valueOf(o).replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n") + "\"";
    }

    private void write(String dir, String name, String text) throws Exception {
        File d = new File(plugin.getDataFolder(), dir);
        d.mkdirs();
        Files.writeString(new File(d, name).toPath(), text, StandardCharsets.UTF_8);
    }
